#### 任务 2.2：统一数据库路径配置（P1-7）

**问题诊断**：
- 根目录 `.env`：`DATABASE_URL="file:./apps/backend/prisma/data/portfolio.db"`
- 后端 `.env`：`DATABASE_URL="file:./prisma/data/portfolio.db"`
- 配置不一致，可能导致开发和打包后使用不同的数据库文件

**解决方案**：
统一使用相对于后端目录的路径，并在 Electron 主进程中动态设置打包后的路径。

**实施步骤**：

##### 步骤 1：统一开发环境配置

1. **修改根目录 `.env`**（从根目录运行后端时）：
   ```bash
   DATABASE_URL="file:./apps/backend/prisma/data/portfolio.db"
   ```

2. **修改 `apps/backend/.env`**（从 backend 目录运行时）：
   ```bash
   DATABASE_URL="file:./prisma/data/portfolio.db"
   ```

3. **在后端添加路径解析逻辑**（`apps/backend/src/config/env.ts`）：
   ```typescript
   import path from 'path';

   export const getDatabaseUrl = () => {
     const dbUrl = process.env.DATABASE_URL || 'file:./prisma/data/portfolio.db';

     // 开发模式：使用配置的路径
     if (process.env.NODE_ENV === 'development') {
       return dbUrl;
     }

     // 生产模式（Electron 打包后）：使用 userData 目录
     if (process.env.ELECTRON_USER_DATA) {
       const userDataPath = process.env.ELECTRON_USER_DATA;
       return `file:${path.join(userDataPath, 'portfolio.db')}`;
     }

     return dbUrl;
   };
   ```

##### 步骤 2：Electron 主进程动态设置路径

编辑 `electron/src/main.ts`：
```typescript
import { app } from 'electron';
import path from 'path';

// 设置数据库路径（打包后使用 userData 目录）
if (app.isPackaged) {
  const userDataPath = app.getPath('userData');
  process.env.ELECTRON_USER_DATA = userDataPath;
  process.env.DATABASE_URL = `file:${path.join(userDataPath, 'portfolio.db')}`;
  console.log('Database path:', process.env.DATABASE_URL);
}

// 启动后端服务（如果集成在 Electron 中）
// ...
```

##### 步骤 3：更新 Prisma 配置

确保 `apps/backend/prisma/schema.prisma` 使用环境变量：
```prisma
datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}
```

**验证方法**：
```bash
# 开发模式测试
cd apps/backend
npm run dev
# 检查日志，确认数据库路径正确

# 打包后测试
# 运行打包的应用，检查 userData 目录是否创建了 portfolio.db
```

**风险评估**：
- **风险**：中等 - 涉及运行时路径解析
- **回退方案**：保持原配置，手动管理数据库文件路径

**预估时间**：15 分钟

**执行结果（2025-11-17）**：
- `apps/backend/src/config/env.ts` 新增 `resolveDatabaseUrl`，打包态会优先落盘到 `userData/portfolio.db` 并写回 `process.env.DATABASE_URL`
- `electron/main.ts` 在 `app.whenReady` 前配置 `ELECTRON_USER_DATA`/`DATABASE_URL`，使 Electron 内嵌后端与 Prisma Client 始终使用同一物理路径

#!/usr/bin/env node
const fs = require('fs');
const path = require('path');
const { execSync } = require('child_process');

console.log('🔍 开始构建前检查...\n');

const checks = [
  {
    name: 'Node.js 版本',
    check: () => {
      const version = process.version;
      const major = parseInt(version.slice(1).split('.')[0]);
      if (major < 18) {
        throw new Error(`需要 Node.js 18+，当前版本：${version}`);
      }
      console.log(`✅ Node.js 版本：${version}`);
    }
  },
  {
    name: 'npm 依赖安装',
    check: () => {
      const hasNodeModules = fs.existsSync(path.join(process.cwd(), 'node_modules'));
      if (!hasNodeModules) {
        throw new Error('node_modules 不存在，请运行 npm install');
      }
      console.log('✅ npm 依赖已安装');
    }
  },
  {
    name: 'Prisma schema',
    check: () => {
      const schemaPath = path.join(process.cwd(), 'apps/backend/prisma/schema.prisma');
      if (!fs.existsSync(schemaPath)) {
        throw new Error('Prisma schema 文件不存在');
      }
      console.log('✅ Prisma schema 存在');
    }
  },
  {
    name: 'TypeScript 配置',
    check: () => {
      const configs = [
        'tsconfig.base.json',
        'packages/domain/tsconfig.json',
        'packages/application/tsconfig.json',
        'packages/infra/tsconfig.json',
        'apps/backend/tsconfig.json',
        'frontend/tsconfig.json',
        'electron/tsconfig.json'
      ];
      configs.forEach(config => {
        const configPath = path.join(process.cwd(), config);
        if (!fs.existsSync(configPath)) {
          throw new Error(`TypeScript 配置缺失：${config}`);
        }
      });
      console.log('✅ TypeScript 配置完整');
    }
  }
];

let failed = false;

checks.forEach(({ name, check }) => {
  try {
    check();
  } catch (error) {
    console.error(`❌ ${name} 检查失败：${error.message}`);
    failed = true;
  }
});

if (failed) {
  console.error('\n❌ 构建前检查失败，请修复上述问题后重试\n');
  process.exit(1);
} else {

}

### C. 常见错误排查

| 错误信息 | 原因 | 解决方案 |
|---------|------|---------|
| `Cannot find module '@uht/domain'` | packages 未构建 | 运行 `npm run build:packages` |
| `@prisma/client not found` | Prisma Client 未生成 | 运行 `npm run build:prisma` |
| `Icon file not found` | 图标文件缺失 | 创建 `electron/assets/icon.*` 或注释 forge.config.js 中的 icon 配置 |
| `electron/renderer/ not found` | 前端未构建 | 运行 `npm run build:frontend` |
| `Database file not found` | 数据库路径错误 | 检查 .env 配置，确保路径正确 |
| `ENOENT: no such file or directory` | 构建产物缺失 | 运行 `npm run build` 完整构建 |
| `Electron Forge config error` | forge.config.js 配置错误 | 检查配置语法，确保路径正确 |
