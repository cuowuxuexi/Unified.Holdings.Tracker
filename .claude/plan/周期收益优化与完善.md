
2. 找到第 386 行：
```typescript

const startKlineDate = formatDate(startDate);
```

```typescript
const totalValueChange = endValue - startValue;
const totalValueChangePercent = startValue > 0 ? totalValueChange / startValue : null;

// 5.2 投资收益率（Modified Dietz）
let periodReturnPercent: number | null = null;
if (startValue > 0) {
  const denominator = startValue + cashFlows * 0.5;
  if (Math.abs(denominator) > 1e-9) {
    periodReturnPercent = (endValue - startValue - cashFlows) / denominator;
  }
}

const periodPnl = periodReturnPercent !== null ? periodReturnPercent * startValue : null;
```

```typescript
const renderPeriodReturn = (
  stats: {
    totalValueChange?: number | null;
    totalValueChangePercent?: number | null;
    periodReturnPercent?: number | null;
    baseDate?: string | null;
    baseDateSource?: string;
    fallbackDays?: number;
  } | null | undefined,
  label: string
) => {
  // 状态1: 数据缺失
  if (!stats || stats.totalValueChangePercent === null || stats.totalValueChangePercent === undefined) {
    return (
      <Tooltip title={`${label}数据暂不可用，可能因为假期、停牌或新上市股票`}>
        <div style={{ textAlign: 'right' }}>
          <div style={{ fontWeight: 700, color: '#999', fontSize: '14px' }}>N/A</div>
          <div style={{ fontSize: '11px', color: '#bbb' }}>数据缺失</div>
        </div>
      </Tooltip>
    );
  }

  // 状态2: 零变化
  if (stats.totalValueChange === 0 && stats.totalValueChangePercent === 0) {
    return (
      <div style={{ textAlign: 'right' }}>
        <div style={{ fontWeight: 700, color: '#888', fontSize: '14px' }}>0.00</div>
        <div style={{ fontSize: '11px', color: '#888' }}>0.00%</div>
      </div>
    );
  }

  // 状态3: 正常涨跌
  const color = (stats.totalValueChange || 0) > 0 ? '#f5222d' : (stats.totalValueChange || 0) < 0 ? '#52c41a' : '#888';
  const prefix = (stats.totalValueChange || 0) > 0 ? '+' : '';

  // 构建 Tooltip 内容
  const tooltipTitle = (
    <div>
      <div>基准日期: {stats.baseDate || 'N/A'}</div>
      <div>数据来源: {getSourceLabel(stats.baseDateSource)}</div>
      {stats.fallbackDays && stats.fallbackDays > 0 && (
        <div>回溯天数: {stats.fallbackDays}天</div>
      )}
      <div style={{ marginTop: 8, borderTop: '1px solid #555', paddingTop: 4 }}>
        <div>总资产变化: 包含存取款影响</div>
        <div>投资收益率: 排除存取款，仅反映投资表现</div>
      </div>
    </div>
  );

  return (
    <Tooltip title={tooltipTitle} placement="left">
      <div style={{ textAlign: 'right' }}>
        {/* 总资产变化 */}
        <div>
          <Text type="secondary" style={{ fontSize: 10 }}>总资产</Text>
          <div style={{ fontWeight: 700, color, fontSize: '14px' }}>
            {prefix}{formatNumber(stats.totalValueChange)} ({prefix}{formatPercent(stats.totalValueChangePercent)})
          </div>
        </div>

        {/* 投资收益率 */}
        <div style={{ marginTop: 2 }}>
          <Text type="secondary" style={{ fontSize: 10 }}>投资收益</Text>
          <div style={{ fontSize: '11px', color: '#888' }}>
            {stats.periodReturnPercent !== null && stats.periodReturnPercent !== undefined
              ? `${(stats.periodReturnPercent || 0) > 0 ? '+' : ''}${formatPercent(stats.periodReturnPercent)}`
              : 'N/A'}
          </div>
        </div>
      </div>
    </Tooltip>
  );
};

// 辅助函数
const getSourceLabel = (source?: string) => {
  switch (source) {
    case 'realtime': return '实时行情';
    case 'kline': return 'K线数据';
    case 'cost': return '成本估值';
    default: return '未知';
  }
};
```
